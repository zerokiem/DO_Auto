import re
from playwright.sync_api import Playwright, sync_playwright, expect


def run(playwright: Playwright) -> None:
    browser = playwright.chromium.launch(headless=False)
    context = browser.new_context()
    page = context.new_page()
    page.goto("https://doffice.npt.com.vn/sign-in?redirectURL=%2Fproject")
    page.get_by_role("textbox", name="Tên đăng nhập").click()
    page.get_by_role("textbox", name="Tên đăng nhập").fill("1npt\\******")
    page.get_by_role("textbox", name="Tên đăng nhập").press("Tab")
    page.get_by_role("textbox", name="Mật khẩu").fill("******")
    page.get_by_role("button", name="Đăng nhập").click()
    page.get_by_role("button", name="Nguyễn Xuân Bình Phòng ban:").click()
    page.get_by_role("menuitem", name="Phó Truyền tải điện Nguyễn Xu").click()

    page.locator("div").filter(has_text="Văn bản").nth(1).click()
    page.get_by_role("link", name="Đã xử lý").click()

    # --- bắt đầu chọn Văn bản thứ 1
    page.locator("tr:nth-child(1) > .mat-cell > div > .vb-item > div > span > .ng-fa-icon").click()
    with page.expect_download() as download_info:
        page.get_by_role("button", name="Tải xuống").click()
    download = download_info.value

    # --- bắt đầu chọn Văn bản thứ 2
    page.locator("div").filter(has_text="Văn bản").nth(1).click()
    page.get_by_role("link", name="Đã xử lý").click()
    page.locator("tr:nth-child(2) > .mat-cell > div > .vb-item > div > span > .ng-fa-icon").click()
    with page.expect_download() as download_info:
        page.get_by_role("button", name="Tải xuống").click()
    download = download_info.value

    # --- các thông tin văn bản: số, ngày, nơi gửi, trích yếu, người chỉ đạo, nội dung chỉ đạo, chủ trì, phối hợp
    page.get_by_text("172/TB-CĐPTC4", exact=True).click()
    page.get_by_text("21/05/2026", exact=True).nth(3).click()
    page.get_by_text("Công ty Truyền tải điện").nth(1).click()
    page.get_by_text("Thông báo Về việc Công đoàn").nth(1).click()
    page.get_by_text("Nguyễn Xuân Bình - 21/05/2026 11:34:45").click()
    page.get_by_text("Chủ trì: CĐ_TTĐTPHCM2").nth(1).click()
    page.get_by_text("Phối hợp: TTH, TĐDĐXO, TĐDCTH, TTBATĐI, TTBABDU").click()

    # ---------------------
    context.close()
    browser.close()


with sync_playwright() as playwright:
    run(playwright)
